// 1. 获取所有收入项 (包含档案固定项 + 专项调整收入)
let b_salary = BASE_SALARY == nil ? 0.0M : BASE_SALARY;
let a_bonus = ATTENDANCE_BONUS == nil ? 0.0M : ATTENDANCE_BONUS;
let k_bonus = KPI_BONUS == nil ? 0.0M : KPI_BONUS;
let h_allow = HOUSING_ALLOW == nil ? 0.0M : HOUSING_ALLOW;
let holiday_bonus = HOLIDAY_BONUS == nil ? 0.0M : HOLIDAY_BONUS; // 动态节日礼金

// 2. 获取所有扣减项 (专项调整扣款)
let late_deduct = LATE_DEDUCTION == nil ? 0.0M : LATE_DEDUCTION;
let util_deduct = DORM_UTILITY == nil ? 0.0M : DORM_UTILITY;   // 动态水电扣款

// 3. 计算【常规】税前收入
let preTaxIncome = (b_salary + a_bonus + k_bonus + h_allow + holiday_bonus) - (late_deduct + util_deduct);

// 4. 年终奖特殊拦截处理 (核心税务策略)
let annual_bonus = ANNUAL_BONUS == nil ? 0.0M : ANNUAL_BONUS;
let annual_tax = 0.0M;

if (annual_bonus > 0.0M) {
    // 假设系统传递了年终奖计税策略：1-单独计税，2-合并计税
    let tax_strategy = ANNUAL_TAX_STRATEGY == nil ? 1 : ANNUAL_TAX_STRATEGY;

    if (tax_strategy == 1) {
        // 【单独计税逻辑】除以12找税率
        let avg_monthly = annual_bonus / 12.0M;
        if (avg_monthly <= 3000.0M) { annual_tax = annual_bonus * 0.03M; }
        elsif (avg_monthly <= 12000.0M) { annual_tax = annual_bonus * 0.1M - 210.0M; }
        // ... (此处省略更高级别阶梯，按法定补充即可) ...
        else { annual_tax = annual_bonus * 0.45M - 15160.0M; }
    } else {
        // 【合并计税逻辑】直接把年终奖砸进当月税前总包
        preTaxIncome = preTaxIncome + annual_bonus;
    }
}

// 5. 常规免税与劳务判断兜底
if (taxRuleCode == 'NO_TAX' || taxRuleCode == 'TAX_FREE') {
    return 0.0M;
}

// 6. 标准综合所得个税计算
let taxable = preTaxIncome - 5000.0M; // 减去免征额
let normal_tax = 0.0M;

if (taxable > 0.0M) {
    if (taxable <= 3000.0M) { normal_tax = taxable * 0.03M; }
    elsif (taxable <= 12000.0M) { normal_tax = taxable * 0.1M - 210.0M; }
    elsif (taxable <= 25000.0M) { normal_tax = taxable * 0.2M - 1410.0M; }
    elsif (taxable <= 35000.0M) { normal_tax = taxable * 0.25M - 2660.0M; }
    elsif (taxable <= 55000.0M) { normal_tax = taxable * 0.3M - 4410.0M; }
    elsif (taxable <= 80000.0M) { normal_tax = taxable * 0.35M - 7160.0M; }
    else { normal_tax = taxable * 0.45M - 15160.0M; }
}

// 7. 返回总税费 = 常规个税 + 年终奖单独计税部分的个税
return normal_tax + annual_tax;
